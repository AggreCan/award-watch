# Award Watch. Every page is open to search engines, in English and in French
# (?lang=fr). The JSON API and the response form are not pages to index.
#
# Crawling the profile and notice pages gains nothing the nightly bulk files
# do not hold: see /llms.txt for the files and their licence. The bulk files
# under /downloads/ may be fetched (ten a minute per address).
User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=no
Allow: /
Disallow: /api/
Disallow: /organizations/*/respond
# Search results: every query would be a page, and each one reads the search
# database. The directory and the sitemaps reach every profile.
Disallow: /search
# Accounts: sign-in, a reader's own pages and the links in their messages.
# Each also says noindex; this keeps crawlers from asking at all.
Disallow: /account
Disallow: /admin
# The analytics views: each reads the records database, they say noindex,
# and the buyer, organization and segment pages they chart stay open.
Disallow: /buyers/*/analytics
Disallow: /organizations/*/analytics
Disallow: /segments/*/analytics
# Their index stays open; a search on it (?q=) reads the search database.
Disallow: /analytics?
Allow: /analytics?lang=fr$
# A filtered tender list, a paged or filtered ranking, a day of new notices
# and a correction form opened from a profile (?organization=) each name the
# plain page as their canonical URL, so crawlers need not walk the query
# forms. The plain pages and their French versions (?lang=fr) stay open.
Disallow: /tenders?
Allow: /tenders?lang=fr$
Disallow: /tenders/new?
Allow: /tenders/new?lang=fr$
Disallow: /rankings?
Allow: /rankings?lang=fr$
Disallow: /corrections?
Allow: /corrections?lang=fr$
Crawl-delay: 5
